CMAKE_MINIMUM_REQUIRED(VERSION 4.1)
PROJECT(genpaylaod
        LANGUAGES C CXX
        VERSION 0.2.0
)

SET(CMAKE_CXX_STANDARD 20)
SET(CMAKE_INSTALL_PREFIX ${CMAKE_CURRENT_SOURCE_DIR}/opt)

INCLUDE(FetchContent)
FETCHCONTENT_DECLARE(CLI11
        URL https://github.com/CLIUtils/CLI11/archive/refs/tags/v2.6.2.tar.gz
        EXCLUDE_FROM_ALL
)
FETCHCONTENT_DECLARE(libFMT
        URL https://github.com/fmtlib/fmt/archive/refs/tags/12.2.0.tar.gz
        EXCLUDE_FROM_ALL
)
FETCHCONTENT_MAKEAVAILABLE(CLI11 libFMT)

ADD_SUBDIRECTORY(app)
