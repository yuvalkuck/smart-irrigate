CMAKE_MINIMUM_REQUIRED(VERSION 4.1)
PROJECT(parent)

SET(CMAKE_CXX_STANDARD 20)
INCLUDE(ExternalProject)
SET(EXT_INSTALL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/opt)
EXTERNALPROJECT_ADD(CLI11
        URL https://github.com/CLIUtils/CLI11/archive/refs/tags/v2.6.2.tar.gz
        INSTALL_DIR ${EXT_INSTALL_DIR}
        CMAKE_ARGS -DCMAKE_INSTALL_PREFIX=${EXT_INSTALL_DIR}
)

EXTERNALPROJECT_ADD(app
        SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/app"
        DEPENDS CLI11
        CMAKE_ARGS -DCMAKE_INSTALL_PREFIX=${EXT_INSTALL_DIR}
        -DCMAKE_PREFIX_PATH=${EXT_INSTALL_DIR}
)
